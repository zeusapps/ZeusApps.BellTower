# Analysis 002 probe scripts. Run from a scratch folder that holds a copy of the reviewed Release build in .\bin
# (src/BellTower/bin/Release/net10.0-windows). Each run uses a fresh isolated data directory and deletes it afterwards.
# Commands used: pwsh -NoProfile -File probe-floor.ps1 -Provider claude|codex -Case tail|before-first-read
#                pwsh -NoProfile -File probe-legacy.ps1 -Provider claude|codex
#                pwsh -NoProfile -File probe-weak.ps1 [-WithTitle]  (unchanged copy of the probe-weak.ps1 section of analysis-001-probe-scripts.ps1.txt)

##### probe-floor.ps1 #####
# Analysis 002 probe: branch-invalidation history floor versus the unpersisted forward cursor, and the floor-0 window.
# Isolated data dir and synthetic transcripts only. Uses the reviewed Release build copied to .\bin.
param([ValidateSet('claude', 'codex')][string]$Provider = 'claude', [ValidateSet('tail', 'before-first-read')][string]$Case = 'tail')
$ErrorActionPreference = 'Stop'
$scratch = Split-Path -Parent $MyInvocation.MyCommand.Path
Add-Type -Path (Join-Path $scratch 'bin\bell-tower.dll')
$root = Join-Path $scratch ("data-floor-" + [guid]::NewGuid().ToString('N').Substring(0, 8))
$work = Join-Path $root 'repo'
New-Item -ItemType Directory -Force (Join-Path $work 'openspec\changes\change-a') | Out-Null
$paths = [BellTower.IO.DataPaths]::new((Join-Path $root 'data'))
$paths.Ensure()
$id = "probe-$Provider"
$slash = $work.Replace('\', '/')
function Ts([int]$s) { ([datetimeoffset]'2026-10-23T12:00:00Z').AddSeconds($s).ToString('yyyy-MM-ddTHH:mm:ss.fffZ') }
function J($o) { $o | ConvertTo-Json -Depth 10 -Compress }
function Chatter([int]$at, [int]$bytes) {
    if ($Provider -eq 'claude') { J @{ isSidechain = $false; type = 'assistant'; timestamp = (Ts $at); sessionId = $id; message = @{ role = 'assistant'; content = @(@{ type = 'text'; text = ('x' * $bytes) }) } } }
    else { J @{ timestamp = (Ts $at); type = 'response_item'; payload = @{ type = 'message'; role = 'assistant'; content = @(@{ type = 'output_text'; text = ('x' * $bytes) }) } } }
}
function WorkA([int]$at) {
    if ($Provider -eq 'claude') {
        J @{ isSidechain = $false; type = 'assistant'; timestamp = (Ts $at); cwd = $slash; sessionId = $id; message = @{ role = 'assistant'; content = @(@{ type = 'tool_use'; id = 'w1'; name = 'Write'; input = @{ file_path = "$slash/openspec/changes/change-a/proposal.md"; content = 'x' } }) } }
        J @{ isSidechain = $false; type = 'user'; timestamp = (Ts ($at + 1)); sessionId = $id; message = @{ role = 'user'; content = @(@{ type = 'tool_result'; tool_use_id = 'w1'; content = 'ok' }) } }
    }
    else {
        J @{ timestamp = (Ts $at); type = 'response_item'; payload = @{ type = 'custom_tool_call'; name = 'apply_patch'; call_id = 'p1'; input = "*** Begin Patch`n*** Add File: $slash/openspec/changes/change-a/proposal.md`n+x`n*** End Patch" } }
        J @{ timestamp = (Ts ($at + 1)); type = 'response_item'; payload = @{ type = 'custom_tool_call_output'; call_id = 'p1'; output = "Exit code: 0`nWall time: 0.1 seconds`nOutput:`nSuccess. Updated the following files:`nA x`n" } }
    }
}
$utf8 = [System.Text.UTF8Encoding]::new($false)
$transcript = Join-Path $root 'transcript.jsonl'
$session = [BellTower.Core.SessionState]::new()
$session.Key = "${Provider}:$id"; $session.Provider = $Provider; $session.SessionId = $id; $session.AttachmentId = "attachment-$id"; $session.Directory = $work
$store = [BellTower.IO.IdentityStore]::new($paths, $null)
$reader = [BellTower.Sessions.IdentityHistoryReader]::new($paths)
function Get-Identity { $store.Get($session.Key, $work, $null, $null, $null, $null) }
function Tick { $n = 0; while ($n -lt 40 -and $reader.Refresh($session)) { $n++ }; $n }
function Live([int]$at) {
    # Same call the live observer makes (OpenSpecSelectionObserver -> WorkPolicy.Apply, Offset -1); the hook arrives after the transcript record.
    $evidence = [BellTower.Core.WorkEvidence]::new('change-a', $work, 'edit', $false, [datetimeoffset]::Parse((Ts $at)), -1)
    $null = $store.Update($session.Key, $work, [Func[BellTower.Core.SessionIdentity, bool]] { param($v) $v.ObservedBranch = 'main'; [BellTower.Core.WorkPolicy]::Apply($v, $evidence) })
}
function Switch-Branch {
    $null = $store.Update($session.Key, $work, [Func[BellTower.Core.SessionIdentity, bool]] { param($v)
        $changed = [BellTower.Core.CheckoutPolicy]::ApplyBranch($v, $work, $work, 'feature')
        $v.ObservedBranch = 'feature'   # the resolver writes ObservedBranch in the same update
        $changed })
}
function Show([string]$label) {
    $i = Get-Identity
    $len = (Get-Item $transcript).Length
    "{0,-28} fileBytes={1} HistoryLength={2} HistoryOffset={3} HistoryFloor={4} ActiveWork={5} WorkSource={6} WorkBranch={7} Subject={8}" -f $label, $len, $i.HistoryLength, $i.HistoryOffset, $i.HistoryFloor, $i.ActiveWork, $i.WorkSource, $i.WorkBranch, $i.Subject
}
"provider=$Provider case=$Case"
# The identity is created here (Update -> GetCore with null initial values); later Get calls only read it.
$null = $store.Update($session.Key, $work, [Func[BellTower.Core.SessionIdentity, bool]] { param($v) [BellTower.Core.HistoryState]::Bind($v, $transcript) })
if ($Case -eq 'tail') {
    [System.IO.File]::WriteAllText($transcript, ((@(Chatter 1 2000; Chatter 2 2000)) -join "`n") + "`n", $utf8)
    "ticks=$(Tick)"; Show 'after first read'
    # Pre-switch work: transcript record at t=10/11, live hook receipt at t=12.
    [System.IO.File]::AppendAllText($transcript, ((@(WorkA 10; Chatter 13 4000)) -join "`n") + "`n", $utf8)
    Live 12
    "ticks=$(Tick)"; Show 'after live work + tail read'
    Switch-Branch; Show 'after branch switch'
    "ticks=$(Tick)"; Show 'after next history pass'
}
else {
    # Invalidation before the first history read (floor 0); the transcript already holds the pre-switch work.
    [System.IO.File]::WriteAllText($transcript, ((@(Chatter 1 2000; WorkA 10; Chatter 13 2000)) -join "`n") + "`n", $utf8)
    Live 12; Show 'after live work (no read yet)'
    Switch-Branch; Show 'after branch switch'
    "ticks=$(Tick)"; Show 'after first history pass'
}
Remove-Item -Recurse -Force $root

##### probe-legacy.ps1 #####
# Analysis 002 probe: a baseline-shaped automatic Change (legacy-selection) after newer verified work and a branch invalidation.
# Isolated data dir only. Uses the reviewed Release build copied to .\bin.
param([ValidateSet('claude', 'codex')][string]$Provider = 'claude')
$ErrorActionPreference = 'Stop'
$scratch = Split-Path -Parent $MyInvocation.MyCommand.Path
Add-Type -Path (Join-Path $scratch 'bin\bell-tower.dll')
$root = Join-Path $scratch ("data-legacy-" + [guid]::NewGuid().ToString('N').Substring(0, 8))
$work = Join-Path $root 'repo'
foreach ($c in 'legacy-x', 'change-a') { New-Item -ItemType Directory -Force (Join-Path $work "openspec\changes\$c") | Out-Null }
$paths = [BellTower.IO.DataPaths]::new((Join-Path $root 'data'))
$paths.Ensure()
$key = "${Provider}:legacy-probe"
$store = [BellTower.IO.IdentityStore]::new($paths, $null)
$null = $store.Get($key, $work, [NullString]::Value, [NullString]::Value, [NullString]::Value, $null)
# Rewrite the file into the 0.1.8 shape: automatic selection wrote Change + SelectionRoot + SelectionEvent + SelectionAt, no marker.
$file = Join-Path $paths.Identities ([BellTower.IO.ContentHash]::Compute($key) + '.json')
$node = [System.Text.Json.Nodes.JsonNode]::Parse([System.IO.File]::ReadAllText($file))
foreach ($n in 'changeSource', 'changeSourceValue') { $null = $node.AsObject().Remove($n) }
$node['change'] = 'legacy-x'; $node['selectionRoot'] = $work; $node['selectionEvent'] = 'evt-1'; $node['selectionAt'] = '2026-10-20T12:00:00+00:00'; $node['futureField'] = 'kept'
[System.IO.File]::WriteAllText($file, $node.ToJsonString())
$presentations = [BellTower.IO.IdentityPresentationStore]::new($paths)
$git = [BellTower.Core.GitContext]::new('main', $null)
function Show([string]$label) {
    $i = $store.Get($key, $work, [NullString]::Value, [NullString]::Value, [NullString]::Value, $null)
    $null = $presentations.Save($i, $git, $null)
    $p = $presentations.Read($key, $null)
    $raw = [System.IO.File]::ReadAllText($file)
    "{0,-34} Change={1} origin={2} ActiveWork={3} WorkSubject={4} Name={5} provenance={6} futureKept={7}" -f $label, $i.Change, [BellTower.Core.ChangeOrigin]::Of($i), $i.ActiveWork, $i.WorkSubject, $i.Name, $(if ($p) { $p.Provenance } else { '(stale)' }), ($raw.IndexOf('"kept"') -ge 0 -and $raw.IndexOf('futureField') -ge 0)
}
"provider=$Provider"
Show 'loaded 0.1.8 record'
$evidence = [BellTower.Core.WorkEvidence]::new('change-a', $work, 'edit', $false, [datetimeoffset]::UtcNow, -1)
$null = $store.Update($key, $work, [Func[BellTower.Core.SessionIdentity, bool]] { param($v) $v.ObservedBranch = 'main'; [BellTower.Core.WorkPolicy]::Apply($v, $evidence) })
Show 'newer verified live work'
$null = $store.Update($key, $work, [Func[BellTower.Core.SessionIdentity, bool]] { param($v) $c = [BellTower.Core.CheckoutPolicy]::ApplyBranch($v, $work, $work, 'feature'); $v.ObservedBranch = 'feature'; $c })
Show 'after branch invalidation'
Remove-Item -Recurse -Force $root
